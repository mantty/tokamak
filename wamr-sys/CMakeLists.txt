cmake_minimum_required(VERSION 3.14)
project(tokamak_wamr C)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(WAMR_ROOT_DIR ${CMAKE_CURRENT_SOURCE_DIR}/vendor)
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 1)
set(WAMR_BUILD_AOT 0)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_FAST_JIT 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_SIMD 1)
set(WAMR_BUILD_TAIL_CALL 1)
set(WAMR_BUILD_EXTENDED_CONST_EXPR 1)
include_directories(${CMAKE_CURRENT_SOURCE_DIR})
set(WAMR_BUILD_REF_TYPES 1)
set(WAMR_BUILD_BULK_MEMORY 1)
set(WAMR_BUILD_MULTI_MODULE 0)
set(WAMR_BUILD_SHARED_MEMORY 0)
set(WAMR_BUILD_SHRUNK_MEMORY 0)
set(WAMR_BUILD_THREAD_MGR 0)
set(WAMR_DISABLE_HW_BOUND_CHECK 1)
set(WAMR_DISABLE_STACK_HW_BOUND_CHECK 1)
set(WAMR_BUILD_CUSTOM_NAME_SECTION 1)
set(WAMR_BUILD_LOAD_CUSTOM_SECTION 1)
set(WAMR_BUILD_INSTRUCTION_METERING 1)
set(WAMR_DISABLE_APP_ENTRY 1)
add_compile_definitions(WASM_API_EXTERN= WASM_RUNTIME_API_EXTERN=)
include(${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake)
add_library(tokamak_wamr STATIC bridge.c ${WAMR_RUNTIME_LIB_SOURCE})
set_target_properties(tokamak_wamr PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
if(MSVC)
    target_compile_options(tokamak_wamr PRIVATE /Gy /Gw)
else()
    target_compile_options(tokamak_wamr PRIVATE -ffunction-sections -fdata-sections -fvisibility=hidden)
endif()
install(TARGETS tokamak_wamr ARCHIVE DESTINATION lib)
